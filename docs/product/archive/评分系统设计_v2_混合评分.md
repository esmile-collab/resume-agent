# 简历评分系统设计 v2 - 混合评分方案

版本：v2.0
日期：2026-03-02
状态：混合策略（规则匹配 + LLM 灵活评分）

---

## 1. 设计目标

1. **平衡客观与主观**：规则匹配处理硬性指标，LLM 处理软性能力
2. **动态权重分配**：根据 JD 参数完整度自适应调整权重
3. **用户友好**：每个分数段都有改进空间，避免"完美简历"的焦虑
4. **可操作性**：建议具体可执行，根据分数动态调整建议深度

def rule_based_scoring(jd_params, resume_params):
    """规则匹配打分，支持归一化"""
    scores = {}
    weights = {
        "education": 0.15,
        "experience": 0.20,
        "skills": 0.25,
        "company": 0.15,
        "industry": 0.10,
        "responsibility": 0.15
    }

    # 只对 JD 中存在的参数进行评分
    if jd_params["education_requirement"]["present"]:
        scores["education"] = calculate_education_match(
            jd_params["education_requirement"]["value"],
            resume_params["education_level"]
        )

    if jd_params["experience_years"]["present"]:
        scores["experience"] = calculate_experience_match(
            jd_params["experience_years"]["value"],
            resume_params["total_years"]
        )

    if jd_params["required_skills"]:
        scores["skills"] = calculate_skill_coverage(
            jd_params["required_skills"],
            resume_params["skills"]
        )

    # ... 其他维度

    # 归一化：按实际参与评分的维度权重
    total_weight = sum(weights[k] for k in scores)
    normalized_score = sum(
        scores[k] * weights[k] / total_weight
        for k in scores
    ) * 100

    return {
        "raw_scores": scores,
        "normalized_score": normalized_score,
        "participating_dimensions": list(scores.keys()),
        "coverage": len(scores) / 6  # 参数完整度
    }

## 5. 分数融合方案

### 5.1 动态权重分配

```python
def calculate_weights(jd_param_completeness):
    """
    根据 JD 参数完整度动态调整权重

    JD 参数越完整 → 规则匹配权重越高（客观指标可靠）
    JD 参数越模糊 → LLM 评分权重越高（依赖主观判断）
    """
    # 参数完整度：0.0 ~ 1.0
    # 规则权重：0.3 ~ 0.5
    rule_weight = 0.3 + (0.2 * jd_param_completeness)
    llm_weight = 1 - rule_weight

    return {
        "rule_weight": rule_weight,  # 0.3 ~ 0.5
        "llm_weight": llm_weight     # 0.7 ~ 0.5
    }

def final_score_calculation(rule_score, llm_score, jd_params):
    """计算最终分数"""
    # 计算参数完整度
    available_dims = count_available_dimensions(jd_params)
    jd_param_completeness = available_dims / 6

    # 动态权重
    weights = calculate_weights(jd_param_completeness)

    # 最终分数
    final_score = (
        rule_score * weights["rule_weight"] +
        llm_score * weights["llm_weight"]
    )

    return {
        "final_score": round(final_score, 1),
        "rule_score": rule_score,
        "llm_score": llm_score,
        "rule_weight": weights["rule_weight"],
        "llm_weight": weights["llm_weight"],
        "jd_param_completeness": jd_param_completeness
    }
```

### 5.2 分数分段定义（用户友好）

| 分数区间 | 匹配度 | 投递建议 | 改进空间说明 |
|---|---|---|---|
| **90-100** | 优秀匹配 | 可以投递，建议微调优化 | 简历已经很棒，微调能让它更出色 |
| **80-89** | 良好匹配 | 可以投递，建议针对性优化 | 基础扎实，有提升空间 |
| **70-79** | 较好匹配 | 建议优化后投递 | 有明显优势，补齐短板后更强 |
| **60-69** | 中等匹配 | 建议重点优化后投递 | 需要针对性改进 |
| **50-59** | 基础匹配 | 谨慎投递，需要较大改进 | 存在明显短板，建议认真优化 |
| **0-49** | 匹配不足 | 不建议投递，建议重新定位 | 差距较大，投递成功率低 |

**设计理念**：
- 即使 90-100 分也建议"微调"，避免用户产生"完美简历"的焦虑
- 每个区间都给出"可以/建议"的正面引导
- 强调"改进空间"而非"缺陷"

## 6. 分层改进建议

### 6.1 建议层级规则

```python
def generate_suggestions(score_card, analysis_result):
    """
    根据分数动态生成建议

    - 高分（≥70分）：当下修改建议（快速见效）
    - 低分（<70分）：当下建议 + 长期提升建议
    """
    score = score_card["final_score"]

    if score >= 70:
        # 中高分：只给出当下修改建议
        return {
            "immediate_actions": extract_immediate_actions(analysis_result),
            "long_term_actions": [],
            "message": "你的简历基础很好，以下建议可以让它更出色："
        }
    else:
        # 低分：给出当下建议 + 长期建议
        return {
            "immediate_actions": extract_immediate_actions(analysis_result),
            "long_term_actions": extract_long_term_actions(analysis_result),
            "message": "建议重点优化以下内容，如果有时间可以长期提升："
        }

def extract_immediate_actions(analysis_result):
    """
    当下修改建议（快速见效）

    特点：
    - 不需要额外时间投入
    - 纯表达层面的优化
    - 立即可以修改
    """
    return [
        {
            "action": "补充具体数据",
            "current": "负责产品优化工作",
            "suggested": "负责产品优化，将用户留存率提升30%",
            "expected_gain": "+3~5分",
            "time_required": "10分钟"
        },
        {
            "action": "强化专业术语",
            "current": "做数据分析",
            "suggested": "使用SQL/Python进行数据分析，输出决策报告",
            "expected_gain": "+2~3分",
            "time_required": "5分钟"
        }
    ]

def extract_long_term_actions(analysis_result):
    """
    长期提升建议（需要时间投入）

    特点：
    - 需要额外学习或项目经验
    - 无法立即完成
    - 可分阶段实施
    """
    return [
        {
            "action": "补充项目经验",
            "suggestion": "可以参与1-2个完整的项目，从0到1的完整经验",
            "expected_gain": "+5~8分",
            "time_required": "1-3个月",
            "priority": "高"
        },
        {
            "action": "考取专业认证",
            "suggestion": "PMP、NPDP等产品经理认证",
            "expected_gain": "+3~5分",
            "time_required": "2-6个月",
            "priority": "中"
        }
    ]
```

## 10. 技术实现要点

### 10.1 LLM 调用优化

```python
async def score_resume(jd_text: str, resume_text: str):
    """混合评分主流程"""

    # 并行调用：参数提取 + 灵活评分
    results = await asyncio.gather(
        llm_extract_params(jd_text, resume_text),
        llm_flexible_scoring(jd_text, resume_text)
    )

    jd_params, resume_params = results[0]
    llm_soft_score = results[1]

    # 规则匹配打分（同步，快速）
    rule_score = rule_based_scoring(jd_params, resume_params)

    # 分数融合
    final = final_score_calculation(
        rule_score["normalized_score"],
        llm_soft_score,
        jd_params
    )

    # 生成分层建议
    suggestions = generate_suggestions(final, llm_soft_score)

    return {
        "score_card": final,
        "rule_details": rule_score,
        "llm_details": llm_soft_score,
        "suggestions": suggestions
    }
```
