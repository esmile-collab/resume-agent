1. 创建数据库表
```python
# src/db/database.py
import sqlite3
from pathlib import Path

def init_db():
    """初始化数据库"""
    db_path = Path(".data/resume_agent.db")
    db_path.parent.mkdir(parents=True, exist_ok=True)

    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()

    # 创建 projects 表
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS projects (
            id TEXT PRIMARY KEY,
            name TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            base_resume_path TEXT
        )
    """)

    conn.commit()
    conn.close()
```

2. 实现 CRUD 操作
```python
# src/db/crud.py
import sqlite3
import uuid
from datetime import datetime
from typing import Optional
from .models import Project
from .database import get_db

class ProjectCRUD:
    def create(self, name: str) -> Project:
        conn = get_db()
        cursor = conn.cursor()

        # 自动生成 8 字符短 ID
        project_id = str(uuid.uuid4())[:8]
        created_at = datetime.now().isoformat()

        cursor.execute(
            "INSERT INTO projects (id, name, created_at) VALUES (?, ?, ?)",
            (project_id, name, created_at)
        )

        conn.commit()
        conn.close()

        return Project(id=project_id, name=name, created_at=created_at)

    def get(self, id: str) -> Optional[Project]:
        conn = get_db()
        cursor = conn.cursor()

        cursor.execute("SELECT * FROM projects WHERE id = ?", (id,))
        row = cursor.fetchone()
        conn.close()

        if row:
            return Project(id=row[0], name=row[1], created_at=row[2])
        return None

    def list_all(self) -> list[Project]:
        conn = get_db()
        cursor = conn.cursor()

        cursor.execute("SELECT * FROM projects")
        rows = cursor.fetchall()
        conn.close()

        return [Project(id=row[0], name=row[1], created_at=row[2]) for row in rows]
```

### M1.5: 最小 CLI

**任务**: 实现 project 命令

**命令接口**:

```bash
resume_agent project init --name <name>
resume_agent project list
resume_agent project get --id <id>
resume_agent project delete --id <id>
```

**实现步骤**:

```python
# src/cli/commands.py
import click
from ..db.crud import ProjectCRUD

@click.command()
@click.option('--name', required=True, help='项目名称')
@click.option('--cycle', default='', help='招聘周期')
def init(name, cycle):
    """创建新项目"""
    crud = ProjectCRUD()
    # 允许同名项目，依靠自动生成的短 ID 区分
    project = crud.create(name, cycle)
    click.echo(f"✓ 项目已创建: {project.id}")

@click.command()
def list():
    """列出所有项目"""
    crud = ProjectCRUD()
    projects = crud.list_all()

    if not projects:
        click.echo("未找到项目")
        return

    click.echo("项目列表:")
    for p in projects:
        click.echo(f"  - {p.id}: {p.name}")

@click.command()
@click.option('--id', required=True, help='项目ID')
def get(id):
    """查询项目"""
    crud = ProjectCRUD()
    project = crud.get(id)

    if not project:
        click.echo(f"✗ 项目不存在: {id}")
        return

    click.echo(f"项目: {project.name}")
    click.echo(f"ID: {project.id}")
    click.echo(f"创建时间: {project.created_at}")

@click.command()
@click.option('--id', required=True, help='项目ID')
def delete(id):
    """删除项目"""
    crud = ProjectCRUD()

    if not crud.get(id):
        click.echo(f"✗ 项目不存在: {id}")
        return

    if crud.delete(id):
        click.echo(f"✓ 项目已删除: {id}")
    else:
        click.echo(f"✗ 删除项目失败: {id}")
```

```python
# src/cli/main.py
import click
from .commands import init, list, get, delete

@click.group()
def cli():
    """Resume Agent CLI"""
    pass

cli.add_command(init)

@click.group()
def project():
    """项目管理命令"""
    pass

project.add_command(init, name="init")
project.add_command(list, name="list")
project.add_command(get, name="get")
project.add_command(delete, name="delete")

cli.add_command(project)
```

```python
import pytest
from src.db.crud import ProjectCRUD

def test_create_project():
    crud = ProjectCRUD()
    project = crud.create("test-project")
    assert project.id is not None
    assert len(project.id) == 8  # 8 字符短 ID
    assert project.name == "test-project"

def test_get_project():
    crud = ProjectCRUD()
    project = crud.create("test-project")
    retrieved = crud.get(project.id)
    assert retrieved is not None
    assert retrieved.id == project.id
```
